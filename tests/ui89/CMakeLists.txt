cmake_minimum_required(VERSION 3.20)
project(KCD2VR_UI89_REGRESSION_TESTS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
# Retain the delivered parent's policy, geometry, queue and actual adapter tests.
add_subdirectory(../ui88 inherited-ui88)
add_executable(ui89_catalogue test_ui_catalogue.cpp)
target_include_directories(ui89_catalogue PRIVATE ../../src)
if(MSVC)
  target_compile_options(ui89_catalogue PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(ui89_catalogue PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME ui89_catalogue COMMAND ui89_catalogue)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME ui89_generated_catalogue COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/generate_ui_catalogue.py --check)
  add_test(NAME ui89_regression_guards COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tools/check_ui89_regressions.py)
endif()
