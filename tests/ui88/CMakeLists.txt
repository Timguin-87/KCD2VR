cmake_minimum_required(VERSION 3.20)
project(KCD2VR_UI88_TESTS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(ui88_policy test_ui_policy.cpp)
target_include_directories(ui88_policy PRIVATE ../../src)
target_link_libraries(ui88_policy PRIVATE Threads::Threads)
if(MSVC)
 target_compile_options(ui88_policy PRIVATE /W4 /UNDEBUG)
else()
 target_compile_options(ui88_policy PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
enable_testing()
add_test(NAME ui88_policy COMMAND ui88_policy)

# Linux/macOS host-only test doubles. These do not certify Windows headers, SEH or detour ABI.
if(NOT WIN32)
 set(STAGE "${CMAKE_CURRENT_BINARY_DIR}/mock_stage")
 file(MAKE_DIRECTORY "${STAGE}")
 foreach(file flashui_probe.h ui_native_canvas.h ui_surface_geometry.h ui_hud_geometry.h ui_menu_input.cpp ui_menu_input.h ui_menu_policy.h diagnostic_trace.h prompt_device_lock.h ui_native_state.cpp ui_native_state.h ui_menu_router.h ui_visibility_registry.h ui_native_catalogue.h)
  configure_file("../../src/${file}" "${STAGE}/${file}" COPYONLY)
 endforeach()
 add_executable(ui88_transport_mock test_ui_transport_mock.cpp)
 target_include_directories(ui88_transport_mock PRIVATE "${STAGE}" "${CMAKE_CURRENT_SOURCE_DIR}/mocks")
 target_compile_options(ui88_transport_mock PRIVATE -Wall -Wextra -Wpedantic)
 add_library(ui88_native_state_mock_syntax OBJECT "${STAGE}/ui_native_state.cpp")
 target_include_directories(ui88_native_state_mock_syntax PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mocks")
 target_compile_options(ui88_native_state_mock_syntax PRIVATE -Wall -Wextra -Wpedantic)
 add_test(NAME ui88_transport_mock COMMAND ui88_transport_mock)
endif()
